cmake_minimum_required(VERSION 3.16)
project(LootApproachSafety LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MOVEMENT_SOURCE "" CACHE FILEPATH "Old movement source for negative control")
set(LOOT_SOURCE "" CACHE FILEPATH "Old opening source for negative control")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" "${MOVEMENT_SOURCE}" "${LOOT_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
 message(FATAL_ERROR "Native approach extraction failed")
endif()
add_executable(loot-approach test.cpp)
target_compile_features(loot-approach PRIVATE cxx_std_17)
target_include_directories(loot-approach PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
foreach(CASE new_threat recovery recheck open_threat manual open_safe legacy short_wait no_move)
 add_test(NAME loot_approach_${CASE} COMMAND loot-approach ${CASE})
endforeach()
