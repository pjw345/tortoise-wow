cmake_minimum_required(VERSION 3.16)
project(ManualCastFeedback LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(SHIM_SOURCE "" CACHE FILEPATH "Old shim source for negative control")
set(ACTION_SOURCE "" CACHE FILEPATH "Old action source for rejection-path control")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" "${SHIM_SOURCE}" "${ACTION_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
 message(FATAL_ERROR "Native cast extraction failed")
endif()
add_executable(cast-feedback test.cpp)
target_compile_features(cast-feedback PRIVATE cxx_std_17)
target_include_directories(cast-feedback PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
foreach(CASE mapping translation reasons unit gameobject unwritten allowed trade)
 add_test(NAME manual_cast_${CASE} COMMAND cast-feedback ${CASE})
endforeach()
