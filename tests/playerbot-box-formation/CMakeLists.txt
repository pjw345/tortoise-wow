cmake_minimum_required(VERSION 3.16)
project(BoxFormation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BOX_SOURCE "" CACHE FILEPATH "Prior production formation for a negative control")
set(OLD_LOADER_SOURCE "" CACHE FILEPATH "Old native loader/command source for a negative control")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${CORE}" "${CMAKE_BINARY_DIR}" "${OLD_LOADER_SOURCE}" "${BOX_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
 message(FATAL_ERROR "Native box/loader extraction failed")
endif()
foreach(VARIANT ZERO ONE TWO MANGOS)
 add_executable(box-${VARIANT} box.cpp)
 target_compile_definitions(box-${VARIANT} PRIVATE MANGOSBOT_${VARIANT})
 target_include_directories(box-${VARIANT} PRIVATE "${CMAKE_BINARY_DIR}")
endforeach()
foreach(CASE party headings stable spacing roster roles tank_assist sizes invalid terrain combat selection help)
 add_test(NAME box_${CASE} COMMAND box-ZERO ${CASE})
endforeach()
foreach(VARIANT ONE TWO MANGOS)
 add_test(NAME box_geometry_${VARIANT} COMMAND box-${VARIANT} headings)
 add_test(NAME box_selection_${VARIANT} COMMAND box-${VARIANT} selection)
endforeach()
