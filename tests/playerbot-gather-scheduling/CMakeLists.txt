cmake_minimum_required(VERSION 3.16)
project(GatherScheduling LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_executable(gather-scheduling scheduling.cpp)
target_compile_features(gather-scheduling PRIVATE cxx_std_17)
target_include_directories(gather-scheduling PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME native_gather_discovery_scheduling COMMAND gather-scheduling)
