cmake_minimum_required(VERSION 3.16)
project(PartyKillLoot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(PACKET_SOURCE "" CACHE FILEPATH "Old registration source for negative control")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${CORE}" "${CMAKE_BINARY_DIR}" "${PACKET_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
 message(FATAL_ERROR "Native kill-loot extraction failed")
endif()
foreach(VARIANT ZERO ONE TWO MANGOS)
 add_executable(kill-${VARIANT} kill.cpp)
 target_compile_definitions(kill-${VARIANT} PRIVATE MANGOSBOT_${VARIANT})
 target_include_directories(kill-${VARIANT} PRIVATE "${CMAKE_BINARY_DIR}")
 add_test(NAME wiring_${VARIANT} COMMAND kill-${VARIANT} wiring)
endforeach()
foreach(CASE valid malformed owner policy profession)
 add_test(NAME kill_${CASE} COMMAND kill-ZERO ${CASE})
endforeach()
