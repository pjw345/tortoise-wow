cmake_minimum_required(VERSION 3.16)
project(PlayerbotShutdownTests LANGUAGES CXX)
enable_testing()
set(CORE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(TRAVEL_ROOT "${CORE_ROOT}/modules/mod-playerbots/src/playerbot")
function(extract source start end output)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(READ "${source}" body)
    string(FIND "${body}" "${start}" first)
    if(first LESS 0)
        message(FATAL_ERROR "Missing native shutdown entry: ${output}")
    endif()
    string(SUBSTRING "${body}" ${first} -1 body)
    string(FIND "${body}" "${end}" length)
    if(length LESS 0)
        message(FATAL_ERROR "Missing native shutdown boundary: ${output}")
    endif()
    string(SUBSTRING "${body}" 0 ${length} fragment)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${fragment}")
endfunction()
extract("${TRAVEL_ROOT}/PlayerbotAI.cpp" "void PlayerbotAI::WaitForTravelSearch()" "void PlayerbotAI::RevalidateMasterPointer()" TravelShutdownWait.inc)
extract("${TRAVEL_ROOT}/PlayerbotScripts.cpp" "void OnShutdown() override" "        // Was World::UpdatePlayerbotsTick(diff)." TravelShutdownHook.inc)
add_executable(TravelShutdownLifecycleTest ../architecture/TravelShutdownLifecycleTest.cpp)
target_compile_features(TravelShutdownLifecycleTest PRIVATE cxx_std_17)
target_include_directories(TravelShutdownLifecycleTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(TravelShutdownLifecycleTest PRIVATE /EHsc)
endif()
add_test(NAME TravelShutdownLifecycleTest COMMAND TravelShutdownLifecycleTest)
set_tests_properties(TravelShutdownLifecycleTest PROPERTIES TIMEOUT 20)

# Preserve the real host ordering: joining only after map unload cannot fix it.
file(READ "${CORE_ROOT}/src/game/World.cpp" world)
string(FIND "${world}" "void World::Shutdown()" shutdown)
string(SUBSTRING "${world}" ${shutdown} -1 world)
string(FIND "${world}" "script->OnShutdown();" hook)
string(FIND "${world}" "sWorld.KickAll();" sessions)
if(hook LESS 0 OR sessions LESS hook)
    message(FATAL_ERROR "Module shutdown must precede session teardown")
endif()
file(READ "${CORE_ROOT}/src/mangosd/WorldRunnable.cpp" runnable)
string(FIND "${runnable}" "sWorld.Shutdown();" shutdown)
string(FIND "${runnable}" "sMapMgr.UnloadAll();" maps)
if(shutdown LESS 0 OR maps LESS shutdown)
    message(FATAL_ERROR "World shutdown must precede terrain/map unload")
endif()
