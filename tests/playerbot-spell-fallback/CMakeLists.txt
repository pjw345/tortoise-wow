cmake_minimum_required(VERSION 3.16)
project(PlayerbotSpellFallback LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../modules/mod-playerbots/src/playerbot/strategy/actions/GenericSpellActions.cpp" source)
string(FIND "${source}" "bool CastSpellAction::isPossible()" start)
string(FIND "${source}" "NextAction** CastSpellAction::getPrerequisites()" end)
if(start LESS 0 OR end LESS start)
 message(FATAL_ERROR "Native spell action extraction boundaries missing")
endif()
math(EXPR length "${end}-${start}")
string(SUBSTRING "${source}" ${start} ${length} methods)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/NativeSpellAction.inc" "${methods}")
add_executable(spell-fallback-test SpellFallbackTest.cpp)
target_compile_features(spell-fallback-test PRIVATE cxx_std_17)
target_include_directories(spell-fallback-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME native_spell_capability_fallback COMMAND spell-fallback-test)
