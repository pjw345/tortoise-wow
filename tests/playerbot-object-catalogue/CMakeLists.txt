cmake_minimum_required(VERSION 3.16)
project(ObjectCatalogue LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}/native-catalogue.inc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(object-catalogue catalogue.cpp)
target_compile_features(object-catalogue PRIVATE cxx_std_17)
target_include_directories(object-catalogue PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME native_custom_object_loot_catalogue COMMAND object-catalogue)
