cmake_minimum_required(VERSION 3.16)
project(PlayerbotLootTargetRecovery LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(TRIGGER_SOURCE "${CORE}/modules/mod-playerbots/src/playerbot/strategy/triggers/LootTriggers.cpp" CACHE FILEPATH "Production trigger source, overridable for the old-code control")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${CORE}" "${CMAKE_BINARY_DIR}" "${TRIGGER_SOURCE}" RESULT_VARIABLE EXTRACT_RESULT)
if(NOT EXTRACT_RESULT EQUAL 0)
  message(FATAL_ERROR "Production trigger/selector extraction failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${TRIGGER_SOURCE}" "${CORE}/modules/mod-playerbots/src/playerbot/strategy/actions/LootAction.cpp")
add_executable(loot-target-recovery recovery.cpp)
target_include_directories(loot-target-recovery PRIVATE "${CMAKE_BINARY_DIR}")
add_test(NAME corpse_movement_range COMMAND loot-target-recovery range)
add_test(NAME blocked_target_reselection COMMAND loot-target-recovery recovery)
