cmake_minimum_required(VERSION 3.16)
project(PlayerbotConfigBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(READ "${CORE}/modules/mod-playerbots/src/playerbot/PlayerbotAIConfig.h" HEADER)
file(READ "${CORE}/modules/mod-playerbots/src/playerbot/PlayerbotAIConfig.cpp" IMPLEMENTATION)
if(HEADER MATCHES "#include [\"<]SystemConfig.h")
  message(FATAL_ERROR "Revision metadata must stay outside the Playerbot config header/PCH")
endif()
string(REGEX MATCH "extern std::string _D_AIPLAYERBOT_CONFIG;" DECLARATION "${HEADER}")
string(REGEX MATCH "#if PLATFORM == PLATFORM_WINDOWS\nstd::string _D_AIPLAYERBOT_CONFIG[^#]+#else\nstd::string _D_AIPLAYERBOT_CONFIG[^#]+#endif" DEFINITION "${IMPLEMENTATION}")
if(NOT DECLARATION OR NOT DEFINITION)
  message(FATAL_ERROR "Cannot extract native config filename declaration/definition")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/config.h" "#pragma once\n#include <string>\n${DECLARATION}\n")
file(WRITE "${CMAKE_BINARY_DIR}/config.cpp" "#include \"config.h\"\n#include \"system.h\"\n${DEFINITION}\n")
file(WRITE "${CMAKE_BINARY_DIR}/system.h" "#pragma once\n#define PLATFORM_WINDOWS 1\n#define SYSCONFDIR \"/etc/turtle/\"\n")
file(WRITE "${CMAKE_BINARY_DIR}/consumer.cpp" "#include \"config.h\"\nstd::string read_config() { return _D_AIPLAYERBOT_CONFIG; }\n")
file(WRITE "${CMAKE_BINARY_DIR}/main.cpp" "#include \"config.h\"\nstd::string read_config();\nint main() { if(read_config()!=EXPECTED) return 1; _D_AIPLAYERBOT_CONFIG=\"changed\"; return read_config()==\"changed\" ? 0 : 2; }\n")
foreach(PLATFORM IN ITEMS 0 1)
  add_executable(config_${PLATFORM} "${CMAKE_BINARY_DIR}/config.cpp" "${CMAKE_BINARY_DIR}/consumer.cpp" "${CMAKE_BINARY_DIR}/main.cpp")
  target_include_directories(config_${PLATFORM} PRIVATE "${CMAKE_BINARY_DIR}")
  if(PLATFORM EQUAL 1)
    target_compile_definitions(config_${PLATFORM} PRIVATE PLATFORM=1 EXPECTED="aiplayerbot.conf")
  else()
    target_compile_definitions(config_${PLATFORM} PRIVATE PLATFORM=0 EXPECTED="/etc/turtle/aiplayerbot.conf")
  endif()
  target_precompile_headers(config_${PLATFORM} PRIVATE "${CMAKE_BINARY_DIR}/config.h")
  add_test(NAME config_${PLATFORM} COMMAND config_${PLATFORM})
endforeach()
