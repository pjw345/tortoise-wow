cmake_minimum_required(VERSION 3.16)
project(PlayerMasterFocusTest LANGUAGES CXX)
enable_testing()
set(CORE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(READ "${CORE_ROOT}/modules/mod-playerbots/src/playerbot/PlayerbotAI.cpp" source)
string(FIND "${source}" "bool PlayerbotAI::ShouldAssistPlayerMaster()" start)
string(FIND "${source}" "bool PlayerbotAI::IsInPvp()" end)
if(start LESS 0 OR end LESS start)
  message(FATAL_ERROR "Native assist policy extraction boundary missing")
endif()
math(EXPR length "${end}-${start}")
string(SUBSTRING "${source}" ${start} ${length} policy)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/NativeMasterFocus.inc" "${policy}")
add_executable(PlayerMasterFocusTest PlayerMasterFocusTest.cpp)
target_compile_features(PlayerMasterFocusTest PRIVATE cxx_std_17)
target_include_directories(PlayerMasterFocusTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
 target_compile_options(PlayerMasterFocusTest PRIVATE /EHsc)
endif()
add_test(NAME PlayerMasterFocusTest COMMAND PlayerMasterFocusTest)
set_tests_properties(PlayerMasterFocusTest PROPERTIES TIMEOUT 20)
