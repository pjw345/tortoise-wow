cmake_minimum_required(VERSION 3.16)
project(ConsoleTeleDelete LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(COMMAND_SOURCE "" CACHE FILEPATH "Old handler source for negative control")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" "${COMMAND_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
 message(FATAL_ERROR "Native handler extraction failed")
endif()
add_executable(tele-delete test.cpp)
target_compile_features(tele-delete PRIVATE cxx_std_17)
target_include_directories(tele-delete PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
foreach(CASE console session missing empty invalid)
 add_test(NAME tele_delete_${CASE} COMMAND tele-delete ${CASE})
endforeach()
