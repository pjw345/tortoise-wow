cmake_minimum_required(VERSION 3.16)
project(PlayerbotConsumables LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
 message(FATAL_ERROR "Consumable extraction failed")
endif()
add_executable(consumables ConsumablesTest.cpp)
target_compile_features(consumables PRIVATE cxx_std_17)
target_compile_definitions(consumables PRIVATE MANGOSBOT_ZERO)
target_include_directories(consumables PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME native_inventory_consumables COMMAND consumables)
