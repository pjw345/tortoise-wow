cmake_minimum_required(VERSION 3.16)
project(ProfessionSkillNotices LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(OLD_PLAYER_SOURCE "" CACHE FILEPATH "Old native UpdateSkillPro for negative control")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${CORE}" "${CMAKE_BINARY_DIR}" "${OLD_PLAYER_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
 message(FATAL_ERROR "Native source extraction failed")
endif()
foreach(MODULE ON OFF)
 foreach(LUA ON OFF)
  set(TARGET "notice-${MODULE}-${LUA}")
  add_executable(${TARGET} notices.cpp)
  target_include_directories(${TARGET} PRIVATE "${CMAKE_BINARY_DIR}")
  if(MODULE)
   target_compile_definitions(${TARGET} PRIVATE TEST_MODULE)
  endif()
  if(LUA)
   target_compile_definitions(${TARGET} PRIVATE ENABLE_ELUNA)
  endif()
  foreach(CASE primary secondary no_gain recipients cap non_profession)
   add_test(NAME "${MODULE}_${LUA}_${CASE}" COMMAND ${TARGET} ${CASE})
  endforeach()
 endforeach()
endforeach()
