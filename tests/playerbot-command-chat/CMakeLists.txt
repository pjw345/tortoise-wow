cmake_minimum_required(VERSION 3.16)
project(PlayerbotCommandChat LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
 message(FATAL_ERROR "Native extraction failed")
endif()
add_executable(command-chat CommandChatTest.cpp)
target_compile_features(command-chat PRIVATE cxx_std_17)
target_include_directories(command-chat PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME native_command_chat_separation COMMAND command-chat)
