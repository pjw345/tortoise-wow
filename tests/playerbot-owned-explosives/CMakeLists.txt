cmake_minimum_required(VERSION 3.16)
project(PlayerbotOwnedExplosives LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
 message(FATAL_ERROR "Native explosive extraction failed")
endif()
foreach(variant ZERO ONE TWO)
 add_executable(explosives_${variant} OwnedExplosivesTest.cpp)
 target_compile_features(explosives_${variant} PRIVATE cxx_std_17)
 target_compile_definitions(explosives_${variant} PRIVATE MANGOSBOT_${variant})
 target_include_directories(explosives_${variant} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 add_test(NAME native_owned_explosives_${variant} COMMAND explosives_${variant})
endforeach()
