cmake_minimum_required(VERSION 3.16)
project(PlayerbotRearCircle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(FORMATION_SOURCE "${CORE}/modules/mod-playerbots/src/playerbot/strategy/values/Formations.cpp" CACHE FILEPATH "Production formation source")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${FORMATION_SOURCE}" "${CMAKE_BINARY_DIR}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
  message(FATAL_ERROR "Production formation extraction failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${FORMATION_SOURCE}")
foreach(VARIANT ZERO ONE TWO)
  add_executable(circle-${VARIANT} circle.cpp)
  target_compile_definitions(circle-${VARIANT} PRIVATE MANGOSBOT_${VARIANT})
  target_include_directories(circle-${VARIANT} PRIVATE "${CMAKE_BINARY_DIR}")
  add_test(NAME rear_circle_${VARIANT} COMMAND circle-${VARIANT})
endforeach()
