cmake_minimum_required(VERSION 3.16)
project(PlayerbotSpeedDiagnosticsTests LANGUAGES CXX)
enable_testing()
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
function(fragment source first last output)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(READ "${source}" body)
    string(FIND "${body}" "${first}" start)
    if(start LESS 0)
        message(FATAL_ERROR "Missing fragment start: ${first}")
    endif()
    string(SUBSTRING "${body}" ${start} -1 body)
    string(FIND "${body}" "${last}" length)
    if(length LESS 0)
        message(FATAL_ERROR "Missing fragment end: ${last}")
    endif()
    string(SUBSTRING "${body}" 0 ${length} body)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${body}")
endfunction()
fragment("${ROOT}/src/game/Objects/Unit.cpp" "void Unit::UpdateSpeed(" "float Unit::GetSpeed(" SpeedRecalc.inc)
fragment("${ROOT}/src/game/Objects/Unit.cpp" "struct SetSpeedRateHelper" "void Unit::SetRooted(" SpeedSetters.inc)
fragment("${ROOT}/modules/mod-playerbots/src/playerbot/BotActionLog.cpp" "void BotActionLog_LogSpeed(Unit*" "void BotActionLog_LogAuraApply(" SpeedLog.inc)
file(READ "${ROOT}/src/shared/BotMovementDiagnostics.h" declarations)
string(REPLACE "#include \"Common.h\"" "" declarations "${declarations}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SpeedDeclarations.inc" "${declarations}")
file(READ "${ROOT}/src/game/PlayerbotStubs.cpp" stubs)
string(REGEX MATCHALL "void BotActionLog_LogSpeed[^\n]+" stubs "${stubs}")
string(REPLACE ";" "\n" stubs "${stubs}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SpeedStubs.inc" "${stubs}")
add_executable(SpeedDiagnosticsTest SpeedDiagnosticsTest.cpp)
target_compile_features(SpeedDiagnosticsTest PRIVATE cxx_std_17)
target_include_directories(SpeedDiagnosticsTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(SpeedDiagnosticsTest PRIVATE /W4 /EHsc)
endif()
add_test(NAME SpeedDiagnosticsTest COMMAND SpeedDiagnosticsTest)
add_executable(NonBotDiagnosticsTest NonBotDiagnosticsTest.cpp)
target_compile_features(NonBotDiagnosticsTest PRIVATE cxx_std_17)
target_include_directories(NonBotDiagnosticsTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME NonBotDiagnosticsTest COMMAND NonBotDiagnosticsTest)
