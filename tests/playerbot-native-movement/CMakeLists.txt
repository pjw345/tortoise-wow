cmake_minimum_required(VERSION 3.16)
project(NativeMovingState LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Native movement extraction failed")
endif()
foreach(variant ZERO ONE TWO MANGOS)
  add_executable(moving-${variant} moving.cpp)
  target_compile_features(moving-${variant} PRIVATE cxx_std_17)
  target_include_directories(moving-${variant} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  if(variant STREQUAL "MANGOS")
    target_compile_definitions(moving-${variant} PRIVATE MANGOS)
  else()
    target_compile_definitions(moving-${variant} PRIVATE CMANGOS MANGOSBOT_${variant})
  endif()
  add_test(NAME native_movement_${variant} COMMAND moving-${variant})
endforeach()
