cmake_minimum_required(VERSION 3.16)
project(PlayerbotWandHandoff LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Native wand handoff extraction failed")
endif()
foreach(variant ZERO ONE TWO)
  add_executable(wand_${variant} WandHandoffTest.cpp)
  target_compile_features(wand_${variant} PRIVATE cxx_std_17)
  target_compile_definitions(wand_${variant} PRIVATE MANGOSBOT_${variant})
  target_include_directories(wand_${variant} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME native_wand_handoff_${variant} COMMAND wand_${variant})
endforeach()
