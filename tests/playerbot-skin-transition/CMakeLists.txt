cmake_minimum_required(VERSION 3.16)
project(PlayerbotSkinTransition LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(OLD_ADD_SOURCE "" CACHE FILEPATH "Override only Add for the old-ignore negative control")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${CORE}" "${CMAKE_BINARY_DIR}" "${OLD_ADD_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
  message(FATAL_ERROR "Native queue extraction failed")
endif()
foreach(VARIANT ZERO ONE TWO MANGOS)
  add_executable(skin-${VARIANT} skin.cpp)
  target_compile_definitions(skin-${VARIANT} PRIVATE MANGOSBOT_${VARIANT})
  target_include_directories(skin-${VARIANT} PRIVATE "${CMAKE_BINARY_DIR}")
endforeach()
add_test(NAME body_to_skin_transition COMMAND skin-ZERO transition)
add_test(NAME skin_retry_and_cleanup COMMAND skin-ZERO controls)
foreach(VARIANT ONE TWO MANGOS)
  add_test(NAME legacy_suppression_${VARIANT} COMMAND skin-${VARIANT} legacy)
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CORE}/modules/mod-playerbots/src/playerbot/LootObjectStack.cpp" "${CORE}/modules/mod-playerbots/src/playerbot/LootObjectStack.h")
