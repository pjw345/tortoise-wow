cmake_minimum_required(VERSION 3.16)
project(MixedCasterGear LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(WEIGHT_SOURCE "" CACHE FILEPATH "Prior native source for negative control")
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/extract.py ${CMAKE_CURRENT_BINARY_DIR} "${WEIGHT_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
 message(FATAL_ERROR "Native scoring extraction failed")
endif()
add_executable(MixedCasterGearTest mixed.cpp)
target_compile_features(MixedCasterGearTest PRIVATE cxx_std_17)
target_include_directories(MixedCasterGearTest PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
foreach(CASE caster_boots plain_int mixed_strength healer pure_physical no_mana attacker statless)
 add_test(NAME mixed_${CASE} COMMAND MixedCasterGearTest ${CASE})
endforeach()
