cmake_minimum_required(VERSION 3.16)
project(PlayerbotLootSafety LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(READ "${CORE}/modules/mod-playerbots/src/playerbot/LootObjectStack.cpp" SOURCE)
string(FIND "${SOURCE}" "bool LootObject::IsApproachSafe(" START)
string(FIND "${SOURCE}" "bool LootObject::IsLootPossible(" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} HELPER)
file(READ "${CORE}/modules/mod-playerbots/src/playerbot/strategy/actions/MovementActions.cpp" SOURCE)
string(FIND "${SOURCE}" "bool MoveToLootAction::Execute(" START)
string(SUBSTRING "${SOURCE}" ${START} -1 SOURCE)
string(FIND "${SOURCE}" "    float dist = sServerFacade.GetDistance2d(bot, wo);" END)
string(SUBSTRING "${SOURCE}" 0 ${END} MOVEMENT)
file(WRITE "${CMAKE_BINARY_DIR}/native-safety.inc" "${HELPER}\n${MOVEMENT}    ++nativeMoves; return true;\n}\n")
file(READ "${CORE}/src/game/Objects/Creature.cpp" AGGRO)
string(FIND "${AGGRO}" "float Creature::GetAttackDistance(" START)
string(FIND "${AGGRO}" "void Creature::SetDeathState(" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${AGGRO}" ${START} ${LENGTH} NATIVE_AGGRO)
file(WRITE "${CMAKE_BINARY_DIR}/native-aggro.inc" "${NATIVE_AGGRO}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CORE}/src/game/Objects/Creature.cpp" "${CORE}/modules/mod-playerbots/src/playerbot/LootObjectStack.cpp")
add_executable(loot-safety safety.cpp)
target_include_directories(loot-safety PRIVATE "${CMAKE_BINARY_DIR}")
add_test(NAME queued_loot_hostile_safety COMMAND loot-safety)
