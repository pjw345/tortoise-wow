cmake_minimum_required(VERSION 3.16)
project(QuestChestAvailability LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
 message(FATAL_ERROR "Native extraction failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS extract.py ../../src/game/LootMgr.cpp ../../src/game/LootMgr.h ../../src/game/Objects/GameObject.cpp ../../src/game/Handlers/LootHandler.cpp ../../modules/mod-playerbots/src/playerbot/LootObjectStack.cpp)
foreach(target QuestChestTest QuestChestBeforeTest)
 add_executable(${target} QuestChestTest.cpp)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 add_test(NAME ${target} COMMAND ${target})
endforeach()
target_compile_definitions(QuestChestBeforeTest PRIVATE BEFORE)
set_tests_properties(QuestChestBeforeTest PROPERTIES WILL_FAIL TRUE)
