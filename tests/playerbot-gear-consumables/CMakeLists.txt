cmake_minimum_required(VERSION 3.16)
project(PlayerbotGearConsumables LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
 message(FATAL_ERROR "Native gear/consumable extraction failed")
endif()
foreach(name Gear Consumables StrategyRestore)
 add_executable(${name}Test ${name}Test.cpp)
 target_compile_features(${name}Test PRIVATE cxx_std_17)
 target_include_directories(${name}Test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 add_test(NAME native_${name} COMMAND ${name}Test)
endforeach()
