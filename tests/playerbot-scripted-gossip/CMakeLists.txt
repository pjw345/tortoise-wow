cmake_minimum_required(VERSION 3.16)
project(PlayerbotScriptedGossip LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Native gossip extraction failed")
endif()
add_executable(scripted-gossip ScriptedGossipTest.cpp)
target_compile_features(scripted-gossip PRIVATE cxx_std_17)
target_compile_definitions(scripted-gossip PRIVATE MANGOSBOT_ZERO)
target_include_directories(scripted-gossip PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME native_scripted_gossip COMMAND scripted-gossip)
