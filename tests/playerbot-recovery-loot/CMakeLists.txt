cmake_minimum_required(VERSION 3.16)
project(PlayerbotRecoveryLoot LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Native recovery/loot method extraction failed")
endif()
add_executable(recovery-loot RecoveryLootTest.cpp)
target_compile_features(recovery-loot PRIVATE cxx_std_17)
target_include_directories(recovery-loot PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME native_recovery_and_loot COMMAND recovery-loot)
add_executable(gathering-pings GatheringPingsTest.cpp)
target_compile_features(gathering-pings PRIVATE cxx_std_17)
target_include_directories(gathering-pings PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME native_gathering_ping_lifecycle COMMAND gathering-pings)
