cmake_minimum_required(VERSION 3.16)
project(UpstreamHealthRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(READ "${CORE}/src/game/Objects/Creature.cpp" SOURCE)
string(FIND "${SOURCE}" "void Creature::SelectLevel(" START)
if(START LESS 0)
  message(FATAL_ERROR "SelectLevel not found")
endif()
string(SUBSTRING "${SOURCE}" ${START} -1 FUNCTION)
string(FIND "${FUNCTION}" "float healthmod = _GetHealthMod(rank);" START)
string(FIND "${FUNCTION}" "// mana" END)
if(START LESS 0 OR END LESS START)
  message(FATAL_ERROR "Cannot locate native SelectLevel health block")
endif()
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${FUNCTION}" ${START} ${LENGTH} HEALTH)
file(WRITE "${CMAKE_BINARY_DIR}/native-health.inc" "${HEALTH}")
add_executable(health-regression health.cpp)
target_include_directories(health-regression PRIVATE "${CMAKE_BINARY_DIR}")
add_test(NAME native_health_scaling COMMAND health-regression)
