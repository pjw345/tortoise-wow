cmake_minimum_required(VERSION 3.16)
project(SkinSkillWarning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(OLD_GATHER_SOURCE "" CACHE FILEPATH "Override only the native gather entrance for a negative control")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${CORE}" "${CMAKE_BINARY_DIR}" "${OLD_GATHER_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
  message(FATAL_ERROR "Native Skinning warning extraction failed")
endif()
foreach(VARIANT ZERO ONE TWO MANGOS)
  add_executable(warning-${VARIANT} warning.cpp)
  target_compile_definitions(warning-${VARIANT} PRIVATE MANGOSBOT_${VARIANT})
  target_include_directories(warning-${VARIANT} PRIVATE "${CMAKE_BINARY_DIR}")
endforeach()
foreach(CASE message repeats ownership readiness range safety silent recovery)
  add_test(NAME warning_${CASE} COMMAND warning-ZERO ${CASE})
endforeach()
foreach(VARIANT ONE TWO MANGOS)
  add_test(NAME warning_legacy_${VARIANT} COMMAND warning-${VARIANT} legacy)
endforeach()
