cmake_minimum_required(VERSION 3.16)
project(WaypointArrivalTest LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/game/Movement/WaypointMovementGenerator.cpp" source)
string(FIND "${source}" "bool WaypointMovementGenerator<Creature>::Update(" start)
string(FIND "${source}" "bool WaypointMovementGenerator<Creature>::GetResetPosition(" end)
if(start LESS 0 OR end LESS start)
 message(FATAL_ERROR "Native waypoint update extraction boundary missing")
endif()
math(EXPR length "${end}-${start}")
string(SUBSTRING "${source}" ${start} ${length} method)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/NativeWaypointUpdate.inc" "${method}")
add_executable(WaypointArrivalTest WaypointArrivalTest.cpp)
target_compile_features(WaypointArrivalTest PRIVATE cxx_std_17)
target_include_directories(WaypointArrivalTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME WaypointArrivalTest COMMAND WaypointArrivalTest)
