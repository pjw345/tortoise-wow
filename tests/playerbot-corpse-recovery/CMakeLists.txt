cmake_minimum_required(VERSION 3.16)
project(PlayerbotCorpseRecovery LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
 message(FATAL_ERROR "Native extraction failed")
endif()
add_executable(corpse-recovery CorpseRecoveryTest.cpp)
target_compile_features(corpse-recovery PRIVATE cxx_std_17)
target_include_directories(corpse-recovery PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME native_safe_party_corpse_recovery COMMAND corpse-recovery)
