cmake_minimum_required(VERSION 3.16)
project(PartyBattleLog LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../src/game/BattleLog.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_attachment.py"
    "${CMAKE_CURRENT_BINARY_DIR}/CastAttachmentMethods.h" RESULT_VARIABLE extraction_result)
if(NOT extraction_result EQUAL 0)
    message(FATAL_ERROR "Could not extract native cast attachment methods")
endif()
add_executable(CastAttachmentTest CastAttachmentTest.cpp)
target_compile_features(CastAttachmentTest PRIVATE cxx_std_17)
target_include_directories(CastAttachmentTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME cast_attachment COMMAND CastAttachmentTest)
add_executable(BattleRecorderTest BattleRecorderTest.cpp)
target_compile_features(BattleRecorderTest PRIVATE cxx_std_17)
target_include_directories(BattleRecorderTest PRIVATE ../../src/game)
target_link_libraries(BattleRecorderTest PRIVATE Threads::Threads)
add_test(NAME battle_recorder COMMAND BattleRecorderTest "${CMAKE_CURRENT_BINARY_DIR}/capture")
add_test(NAME battle_reports COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_reports.py" "${CMAKE_CURRENT_BINARY_DIR}/capture")
set_tests_properties(battle_recorder PROPERTIES FIXTURES_SETUP captured_battles)
set_tests_properties(battle_reports PROPERTIES FIXTURES_REQUIRED captured_battles)

add_executable(ObservationTest ObservationTest.cpp)
target_compile_features(ObservationTest PRIVATE cxx_std_17)
target_include_directories(ObservationTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME near_teleport_observation COMMAND ObservationTest)
