cmake_minimum_required(VERSION 3.16)
project(NontradedItemOffer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(TRADE_SOURCE "" CACHE FILEPATH "Production TradeAction override for old-code controls")
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${CORE}" "${CMAKE_BINARY_DIR}" "${TRADE_SOURCE}" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
 message(FATAL_ERROR "Native trade extraction failed")
endif()
foreach(VARIANT ZERO ONE TWO MANGOS)
 add_executable(trade-${VARIANT} trade.cpp)
 target_compile_definitions(trade-${VARIANT} PRIVATE MANGOSBOT_${VARIANT})
 target_include_directories(trade-${VARIANT} PRIVATE "${CMAKE_BINARY_DIR}")
endforeach()
foreach(CASE first_nt explicit_nt soulbound repeat occupied denied partner native_reject normal money missing script_change cancelled)
 add_test(NAME trade_${CASE} COMMAND trade-ZERO ${CASE})
endforeach()
foreach(VARIANT ONE TWO MANGOS)
 add_test(NAME trade_legacy_${VARIANT} COMMAND trade-${VARIANT} legacy)
 add_test(NAME trade_slot_${VARIANT} COMMAND trade-${VARIANT} explicit_nt)
endforeach()
